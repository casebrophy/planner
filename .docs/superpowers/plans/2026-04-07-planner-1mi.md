## Task 3: Create useTaskNotes Composable

**Files:**
- Create: `api/services/frontend/web/src/composables/useTaskNotes.ts`

The composable loads notes filtered by taskId (similar to `useTaskDetail` which loads a single task). It returns reactive notes array and CRUD methods.

- [ ] **Step 1: Write useTaskNotes composable**

```typescript
import { onMounted, computed, ref, Ref } from 'vue'
import { useNoteStore } from '@/stores/noteStore'
import { storeToRefs } from 'pinia'
import type { NewNote, UpdateNote } from '@/types'

export function useTaskNotes(taskId: string | Ref<string>) {
  const noteStore = useNoteStore()
  const { items, loading } = storeToRefs(noteStore)

  // Resolve taskId whether it's a string or Ref
  const resolvedTaskId = typeof taskId === 'string' ? taskId : taskId.value

  // Filter items to only those with matching taskId
  const notes = computed(() => {
    if (!resolvedTaskId) return []
    return items.value.filter((note) => note.taskId === resolvedTaskId)
  })

  async function load() {
    // Load notes filtered by taskId using the noteService list method
    // The noteService CRUD factory supports filtering via the filter object
    await noteStore.fetchList({ taskId: resolvedTaskId })
  }

  async function addNote(data: NewNote) {
    // Create a new note with taskId included
    const noteWithTask: NewNote = {
      ...data,
      taskId: resolvedTaskId,
    }
    return noteStore.create(noteWithTask)
  }

  async function updateNote(noteId: string, data: UpdateNote) {
    return noteStore.update(noteId, data)
  }

  async function deleteNote(noteId: string) {
    return noteStore.remove(noteId)
  }

  onMounted(load)

  return {
    notes,
    loading,
    addNote,
    updateNote,
    deleteNote,
    reload: load,
  }
}
```

async function handleAddNote(data: NewNote | UpdateNote) {
  if (editingNote.value) {
    await updateNote(editingNote.value.id, data as UpdateNote)
    editingNote.value = null
  } else {

  if (confirmDeleteNote.value) {

    confirmDeleteNote.value = null
  }
}

  showNoteForm.value = true
